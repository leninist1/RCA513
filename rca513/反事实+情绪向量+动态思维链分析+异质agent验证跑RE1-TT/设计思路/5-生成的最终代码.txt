### `src/operators.py`

这里我们不重新实现您的 RCA 算法，而是定义算子**配置结构**，最终让您的算法根据这些配置来执行。您已有的 RCA 算法会接收一个算子配置字典，并据此调整行为。

```python
from typing import Dict, Any

class OperatorConfig:
    """算子配置容器，您的 RCA 算法根据它来调整行为"""
    def __init__(self, op_id: str, name: str, params: Dict = None):
        self.id = op_id
        self.name = name
        self.params = params or {}

    def to_dict(self):
        return {"id": self.id, "name": self.name, "params": self.params}

def load_operators_from_yaml(filepath: str) -> list:
    import yaml
    with open(filepath, 'r') as f:
        config = yaml.safe_load(f)
    ops = []
    for op in config['operators']:
        ops.append(OperatorConfig(str(op['id']), op['name'], op.get('params', {})))
    return ops
```

### `src/rca_system.py`

这个模块是**您已有的 RCA 算法的外壳**。您需要实现一个函数 `run_rca_with_config(case_data: dict, operator_list: list) -> dict`，它接收 case 数据和算子配置列表，返回预测根因和推理过程。

```python
from typing import List, Dict
from src.operators import OperatorConfig

class RCASystem:
    """
    封装您已有的 RCA 算法。
    您需要在 .predict() 中调用您现有的算法代码。
    """
    def __init__(self, operators: List[OperatorConfig]):
        self.operators = operators

    def predict(self, case_data: Dict) -> Dict:
        """
        返回：{
            "root_cause": "service-4",
            "reasoning_trace": "...",       # 可选，评估时需要
            "intermediate_scores": {...}    # 可选
        }
        """
        # 这里是您已有算法的入口，根据 self.operators 调整参数
        from your_rca_module import run_rca_with_operators   # 替换为实际导入
        return run_rca_with_operators(case_data, self.operators)
```

**您需要做的**：将 `your_rca_module` 替换为您的实际 RCA 算法模块，并实现 `run_rca_with_operators(case_data, operators)`，它根据 operators 的配置（例如 scoring 权重、是否反事实等）来执行推理。

如果您的算法暂时无法动态调整，可以先使用固定的算子配置，后续再扩展。

### `src/evaluator.py`

此评估器调用 `RCASystem` 在 30 个训练 case 和 5 个 holdout case 上运行，并计算多维指标。

```python
import json
import os
import random
from typing import Dict, List, Tuple
from src.variant import Variant
from src.rca_system import RCASystem
from src.operators import OperatorConfig, load_operators_from_yaml

class Evaluator:
    def __init__(self, eval_config: dict):
        self.weights = eval_config['weights']
        self.train_dir = eval_config['train_cases_dir']
        self.holdout_dir = eval_config['holdout_cases_dir']
        self.annotation_file = eval_config['annotation_file']
        self.lambda_r = eval_config.get('lambda_r', 0.1)

        # 加载标注
        with open(self.annotation_file) as f:
            self.ground_truth = json.load(f)  # 格式: {case_id: "service-4"}

        # 加载 case 文件列表（假设为 JSON 文件）
        self.train_cases = self._load_cases(self.train_dir)
        self.holdout_cases = self._load_cases(self.holdout_dir)

    def _load_cases(self, directory: str) -> List[Dict]:
        cases = []
        for fname in sorted(os.listdir(directory)):
            if fname.endswith('.json'):
                with open(os.path.join(directory, fname)) as f:
                    case_data = json.load(f)
                    case_data['case_id'] = fname.replace('.json', '')
                    cases.append(case_data)
        return cases

    def _build_rca(self, variant: Variant, base_operators: List[OperatorConfig]) -> RCASystem:
        """
        根据变体修改算子配置，构建新的 RCASystem。
        这里仅演示简单修改：如果 target_operator 是已有算子，则更新其参数；
        如果是新的（如 1.5），则插入一个新算子。
        实际对接您的算法时，您需要更精细的映射。
        """
        new_ops = [OperatorConfig(op.id, op.name, op.params.copy()) for op in base_operators]
        target = variant.target_operator
        if '.' not in target:  # 修改已有算子
            for op in new_ops:
                if op.id == target:
                    # 根据 variant.modification 更新参数 (这里简单示例)
                    if 'new_params' in variant.modification:
                        op.params.update(variant.modification['new_params'])
                    break
        else:  # 插入新算子（这里假设插入一个 dummy 算子）
            insert_pos = int(target.split('.')[0])
            new_op = OperatorConfig(target, "custom_step", variant.modification.get('params', {}))
            new_ops.insert(insert_pos, new_op)
        return RCASystem(new_ops)

    def _run_accuracy(self, rca: RCASystem, cases: List[Dict]) -> Tuple[float, List[Dict]]:
        correct = 0
        traces = []
        for case in cases:
            result = rca.predict(case)
            predicted = result.get('root_cause')
            true = self.ground_truth.get(case['case_id'])
            if predicted == true:
                correct += 1
            traces.append({
                'case_id': case['case_id'],
                'predicted': predicted,
                'true': true,
                'reasoning': result.get('reasoning_trace', '')
            })
        accuracy = correct / len(cases) if cases else 0
        return accuracy, traces

    def _compute_stability(self, rca: RCASystem, n_repeats=3) -> float:
        # 选取前 5 个 case 重复运行，看结果一致性
        test_cases = self.train_cases[:5]
        consistent = 0
        for case in test_cases:
            results = [rca.predict(case)['root_cause'] for _ in range(n_repeats)]
            if len(set(results)) == 1:
                consistent += 1
        return consistent / len(test_cases) if test_cases else 0

    def _calc_complexity_penalty(self, variant: Variant, champion: Variant) -> float:
        # 简单用 modification 的字数增加比例
        base_len = len(str(champion.modification)) if champion.modification else 100
        new_len = len(str(variant.modification)) if variant.modification else 0
        increase = max(0, (new_len - base_len) / max(base_len, 1))
        # 每增加 20% 复杂度扣 1%，最大扣 0.5
        penalty = min(0.5, increase / 0.2 * 0.01)
        return penalty

    def _run_ablation(self, variant: Variant, rca: RCASystem) -> float:
        # 消融：删除 variant 引入的修改，回退到 champion 算子，看准确率下降
        # 这里需要 champion 构建的 baseline 系统
        # 为简化，返回随机值，实际应实现
        return 0.8

    def _compute_emotion_consistency(self, variant: Variant, agent_state: dict, traces: List) -> float:
        u = agent_state.get('u', 0.5)
        s = agent_state.get('s', 0)
        e = agent_state.get('e', 0.5)
        # 动作分类：根据 target_operator 和 modification 中的关键词简单判断
        action = "exploration"  # 默认探索
        if "threshold" in str(variant.modification) or "fine-tune" in str(variant.modification):
            action = "convergence"
        if "refactor" in str(variant.modification) or "replace" in str(variant.modification):
            action = "paradigm_jump"

        reward = 0.0
        if u > 0.6 and action == "exploration":
            reward += 1.0
        elif u > 0.6 and action == "convergence":
            reward -= 0.5
        if s >= 3 and action == "paradigm_jump":
            reward += 1.5
        elif s >= 3 and action == "convergence":
            reward -= 1.0
        # 简单返回
        return reward

    def evaluate(self, variant: Variant, champion: Variant, agent_state: dict, base_operators: List[OperatorConfig]) -> Tuple[Dict, List]:
        rca = self._build_rca(variant, base_operators)
        champ_rca = self._build_rca(champion, base_operators)

        # 1. 准确率 (训练集)
        train_acc, train_traces = self._run_accuracy(rca, self.train_cases)
        # 2. 稳定性
        stability = self._compute_stability(rca)
        # 3. 泛化 (留出集)
        holdout_acc, _ = self._run_accuracy(rca, self.holdout_cases)
        # 4. 复杂度惩罚
        complexity_penalty = self._calc_complexity_penalty(variant, champion)
        # 5. 消融测试
        ablation_score = self._run_ablation(variant, rca)  # 简化

        R_ext = (train_acc * self.weights['accuracy'] +
                 stability * self.weights['stability'] +
                 holdout_acc * self.weights['generalization'] +
                 (1 - complexity_penalty) * self.weights['complexity_penalty'] +
                 ablation_score * self.weights['ablation_survival'])

        R_int = self._compute_emotion_consistency(variant, agent_state, train_traces)
        R_total = R_ext + self.lambda_r * R_int

        metrics = {
            'accuracy': train_acc,
            'stability': stability,
            'generalization': holdout_acc,
            'complexity_penalty': complexity_penalty,
            'ablation_survival': ablation_score,
            'R_ext': R_ext,
            'R_int': R_int,
            'R_total': R_total
        }
        return metrics, train_traces
```

```python
import os
import abc
import json
import anthropic
import google.generativeai as genai

class BaseLLMClient(abc.ABC):
    @abc.abstractmethod
    def generate(self, prompt: str, system_prompt: str = "") -> str:
        pass

class ClaudeClient(BaseLLMClient):
    def __init__(self, model: str, api_key_env: str):
        self.model = model
        api_key = os.environ.get(api_key_env)
        if not api_key:
            raise ValueError(f"环境变量 {api_key_env} 未设置")
        self.client = anthropic.Anthropic(api_key=api_key)

    def generate(self, prompt: str, system_prompt: str = "") -> str:
        response = self.client.messages.create(
            model=self.model,
            max_tokens=4096,
            system=system_prompt,
            messages=[{"role": "user", "content": prompt}]
        )
        return response.content[0].text

class GeminiClient(BaseLLMClient):
    def __init__(self, model: str, api_key_env: str):
        self.model = model
        api_key = os.environ.get(api_key_env)
        if not api_key:
            raise ValueError(f"环境变量 {api_key_env} 未设置")
        genai.configure(api_key=api_key)
        self.gemini_model = genai.GenerativeModel(model)

    def generate(self, prompt: str, system_prompt: str = "") -> str:
        # Gemini 没有独立的 system prompt，合并到 user prompt
        full_prompt = f"{system_prompt}\n\n{prompt}" if system_prompt else prompt
        response = self.gemini_model.generate_content(full_prompt)
        return response.text

class MockLLMClient(BaseLLMClient):
    """用于测试，返回固定变体建议"""
    def __init__(self, agent_id: str):
        self.agent_id = agent_id

    def generate(self, prompt: str, system_prompt: str = "") -> str:
        # 返回一个符合格式的 JSON 样例
        if "claude" in self.agent_id:
            return json.dumps({
                "variant_id": f"V7.1-{self.agent_id}",
                "parent_version": "V7.0",
                "target_operator": "1",
                "modification": {
                    "what_changes": "调整时间权重",
                    "new_params": {"weight_time": 0.5}
                }
            })
        else:
            return json.dumps({
                "variant_id": f"V7.1-{self.agent_id}",
                "parent_version": "V7.0",
                "target_operator": "1",
                "modification": {
                    "what_changes": "引入拓扑指纹特征",
                    "new_params": {"use_fingerprint": True}
                }
            })

def create_llm_client(agent_cfg: dict, mock: bool = False):
    if mock:
        return MockLLMClient(agent_cfg['llm_client'])
    client_type = agent_cfg['llm_client']
    model = agent_cfg.get('model', '')
```

```python
from difflib import SequenceMatcher
from typing import Dict, List
from src.variant import Variant

class InternalStateTracker:
    def __init__(self):
        self.agent_history = {}  # agent_id -> list of (variant, state, metrics)

    def update(self, agent_id: str, old_state: dict, variant: Variant,
               metrics: dict, reasoning_traces: List[Dict]) -> dict:
        # 简单规则实现；后续可增强
        u = self._calc_uncertainty(reasoning_traces)
        c = self._extract_confidence(reasoning_traces)
        s = self._update_fatigue(agent_id, variant)
        e = self._update_exploration(agent_id, metrics, variant)

        new_state = {"u": u, "c": c, "s": s, "e": e}
        # 记录历史
        self.agent_history.setdefault(agent_id, []).append(
            (variant.variant_id, new_state, metrics)
        )
        return new_state

    def _calc_uncertainty(self, traces: List[Dict]) -> float:
        # 简单统计推理文本中的不确定词
        keywords = ['possibly', 'might', 'unclear', '不确定', '可能']
        matches = 0
        for t in traces:
            reasoning = t.get('reasoning', '')
            for kw in keywords:
                matches += reasoning.lower().count(kw)
        # 映射到 0~1，假设最多出现20次
        return min(1.0, matches / 20.0)

    def _extract_confidence(self, traces: List[Dict]) -> float:
        # 假设推理文本中有 "置信度:0.8" 之类的标记，这里返回默认
        return 0.5

    def _update_fatigue(self, agent_id: str, current_variant: Variant) -> float:
        if agent_id not in self.agent_history or not self.agent_history[agent_id]:
            return 0
        # 取最近两次变体的 modification 文本相似度
        last_two = self.agent_history[agent_id][-2:]  # 获取最近两次的variant_id
        # 简化：直接从历史记录中找对应 variant 的 modification
        # 由于我们存储的是 (variant_id, state, metrics)，modification需要从全局获取
        # 这里不实现复杂的相似计算，返回默认
        return 0

    def _update_exploration(self, agent_id: str, metrics: dict, variant: Variant) -> float:
        # 若 Type C 准确率低，则冲动升高
        # 目前指标里没有 Type C 细分，假设从 metrics 中获取
        type_c_acc = metrics.get('type_c_accuracy', 0.0)
        if type_c_acc < 0.5:
            return min(1.0, (1 - type_c_acc) * 2)
        return 0.0
```

```python
from collections import deque
from typing import Dict

class FailureMemory:
    def __init__(self, capacity=20):
        self.memory = deque(maxlen=capacity)

    def add(self, entry: Dict):
        self.memory.append(entry)

    def summarize(self, philosophy: str) -> str:
        relevant = [e for e in self.memory if e.get('philosophy') == philosophy]
        if not relevant:
            return "暂无相关失败记录。"
        summary = "近期失败教训：\n"
        for i, entry in enumerate(relevant[-3:]):
            summary += f"{i+1}. 故障类型 {entry.get('fault_type','?')}: {entry.get('reason','')}\n"
        return summary
```

```python
from typing import List
from src.variant import Variant
import random

class Selector:
    def __init__(self, k_top=3):
        self.k = k_top

    def select(self, variants: List[Variant], champion: Variant) -> List[Variant]:
        # 按 R_total 排序
        sorted_vars = sorted(variants, key=lambda v: v.metrics.get('R_total', 0), reverse=True)
        if not sorted_vars:
            return []
        selected = [sorted_vars[0]]

        # 多样性保留：选择与冠军编辑距离最大的（这里用 variant_id 的简单差异）
        # 实际可基于 modification 文本的相似度
        champion_desc = str(champion.modification)
        best_diversity = None
        best_dist = -1
        for v in sorted_vars[1:]:
            d = self._edit_distance(str(v.modification), champion_desc)
            if d > best_dist and d > 10:  # 阈值
                best_dist = d
                best_diversity = v
        if best_diversity and best_diversity not in selected:
            selected.append(best_diversity)

        # 局部最优：选 Type C 准确率最高的（假设 metrics 中有 type_c_accuracy）
        type_c_best = max(variants, key=lambda v: v.metrics.get('type_c_accuracy', 0))
        if type_c_best not in selected:
            selected.append(type_c_best)

        # 补充到 k 个
        for v in sorted_vars:
            if len(selected) >= self.k:
                break
            if v not in selected:
                selected.append(v)
        return selected[:self.k]

    def _edit_distance(self, a: str, b: str) -> int:
        # 简单的 Levenshtein 距离，可使用外部库；这里用内置序列匹配器近似
        from difflib import SequenceMatcher
        return int((1 - SequenceMatcher(None, a, b).ratio()) * max(len(a), len(b)))
```

```python
from src.variant import Variant

def build_agent_prompt(agent_cfg: dict, champion: Variant, failure_summary: str, gen: int, eval_results_str: str) -> str:
```

### `src/evolution_manager.py`

进化主循环，整合所有模块。

```python
import yaml
import json
import os
from datetime import datetime
from src.operators import load_operators_from_yaml, OperatorConfig
from src.variant import Variant
from src.evaluator import Evaluator
from src.selector import Selector
from src.internal_state_tracker import InternalStateTracker
from src.failure_memory import FailureMemory
from src.prompt_builder import build_agent_prompt
from src.rca_system import RCASystem
from src.llm_clients import create_llm_client

class EvolutionManager:
    def __init__(self, config_dir="configs", mock_llm=True):
        # 加载配置
        with open(os.path.join(config_dir, 'agents.yaml'), 'r') as f:
            self.agents_cfg = yaml.safe_load(f)['agents']
        with open(os.path.join(config_dir, 'evaluation.yaml'), 'r') as f:
            self.eval_cfg = yaml.safe_load(f)
        with open(os.path.join(config_dir, 'operators.yaml'), 'r') as f:
            self.operators_cfg = f  # 先不读取，用 load_operators

        self.base_operators: List[OperatorConfig] = load_operators_from_yaml(
            os.path.join(config_dir, 'operators.yaml'))

        # 初始化 LLM 客户端
        self.llm_clients = {}
        for agent_id, cfg in self.agents_cfg.items():
            self.llm_clients[agent_id] = create_llm_client(cfg, mock=mock_llm)

        # 初始化其他组件
        self.evaluator = Evaluator(self.eval_cfg)
        self.selector = Selector(k_top=3)
        self.tracker = InternalStateTracker()
        self.failure_memory = FailureMemory()

        # 初始 Champion: 从算子配置构建
        self.champion = Variant(
            variant_id="V7.0",
            parent_version="base",
            target_operator="",
            modification={"what_changes": "初始基础版本"},
            proposer_agent="system",
            proposer_internal_state={}
        )
        # 给 champion 填充初始指标（实际可先跑一次评估）
        self.champion.metrics = {
            'accuracy': 0.733,
            'R_total': 0.733
        }

        self.current_gen = 0
        self.generations = self.eval_cfg.get('generations', 10)
        self.log_dir = "logs"
        self.ckpt_dir = "checkpoints"
        os.makedirs(self.log_dir, exist_ok=True)
        os.makedirs(self.ckpt_dir, exist_ok=True)

    def run(self):
        for gen in range(self.generations):
            self.current_gen = gen
            print(f"\n=== 进化第 {gen} 代 ===")
            variants = self._run_one_generation()
            # 存储日志
            self._log(gen, variants)
            self._checkpoint(gen)

    def _run_one_generation(self) -> List[Variant]:
        variants = []
        for agent_id, cfg in self.agents_cfg.items():
            # 获取失败记忆摘要
            failure_summary = self.failure_memory.summarize(cfg['attribution_philosophy'])
            # 构建评估结果摘要
            eval_str = f"总准确率={self.champion.metrics.get('accuracy',0):.2f}"
            # 构建 prompt
            prompt = build_agent_prompt(cfg, self.champion, failure_summary, self.current_gen, eval_str)
            # 调用 LLM
            client = self.llm_clients[agent_id]
            try:
                raw_response = client.generate(prompt)
                proposal = json.loads(raw_response)
            except Exception as e:
                print(f"  LLM 调用或解析失败 ({agent_id}): {e}")
                continue

            # 创建变体
            variant = Variant.from_proposal(proposal, agent_id, cfg['internal_state'])
            print(f"  收到变体: {variant.variant_id} (修改算子 {variant.target_operator})")

            # 评估变体
            metrics, traces = self.evaluator.evaluate(
                variant, self.champion, cfg['internal_state'], self.base_operators
            )
            variant.metrics = metrics
            # 更新内部状态
            new_state = self.tracker.update(agent_id, cfg['internal_state'], variant, metrics, traces)
            cfg['internal_state'] = new_state

            # 记录失败
            if metrics['accuracy'] < 0.5:
                self.failure_memory.add({
                    'philosophy': cfg['attribution_philosophy'],
                    'fault_type': 'Type C',
                    'reason': f"准确率仅 {metrics['accuracy']:.2f}"
                })

            variants.append(variant)

        # 选择下一代 parents
        parents = self.selector.select(variants, self.champion)
        if parents:
            # 更新 champion 为 R_total 最高的
            new_champion = max(parents, key=lambda v: v.metrics.get('R_total', 0))
            self.champion = new_champion
            print(f"  新 Champion: {self.champion.variant_id}, R_total={self.champion.metrics.get('R_total',0):.3f}")

        return variants

    def _log(self, gen, variants):
        log_path = os.path.join(self.log_dir, f"gen_{gen:02d}.json")
        log_data = {
            "generation": gen,
            "champion": self.champion.variant_id,
            "variants": [{
                "id": v.variant_id,
                "agent": v.proposer_agent,
                "target": v.target_operator,
                "metrics": v.metrics
            } for v in variants]
        }
        with open(log_path, 'w', encoding='utf-8') as f:
            json.dump(log_data, f, indent=2, ensure_ascii=False)

    def _checkpoint(self, gen):
        # 保存完整状态以便断点续跑 (可扩展)
        pass
```

## 3. 如何与您的数据及 RCA 算法集成

1. **数据准备**：将您的 case 文件放入 `data/cases/`（训练）和 `data/holdout_cases/`（留出），每个 case 一个 JSON，包含您的特征数据（如时间序列、拓扑等）。标注文件 `data/annotations/ground_truth.json` 格式为 `{"case_001": "service-A", ...}`。

2. **RCA 算法对接**：在 `src/rca_system.py` 中，将 `from your_rca_module import run_rca_with_operators` 替换成您实际的 RCA 函数。您的函数需要接受两个参数：`case_data`（字典，包含 case 所有原始信息）和 `operators`（OperatorConfig 列表）。它返回 `dict`，至少包含 `root_cause` 键。

3. **算子参数调整**：我们的变体修改目前只演示了更新已有算子的参数（通过 `new_params`）和插入自定义步骤。您需要根据您的算法能力扩展 `Evaluator._build_rca()` 方法，让它可以实际影响您的算法行为。

4. **API 密钥**：在环境变量中设置 `ANTHROPIC_API_KEY` 和 `GEMINI_API_KEY`。代码会从 `agents.yaml` 中指定的 `api_key_env` 读取对应变量名。运行时不要使用 `--mock`，即可调用真实 LLM。
